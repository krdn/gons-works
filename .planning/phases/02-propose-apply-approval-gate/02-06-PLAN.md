       interface PendingMarker { nonce, stack, command, service, fileEdit?, reasoning, user_prompt, sha_before, ts_created, docker_started_at, docker_finished_at, exit_code }

       async function writeMarker(marker: PendingMarker): Promise<void> { writeFileSync(`state/.pending/${marker.nonce}.json`, JSON.stringify(marker, null, 2)) }
       async function updateMarker(nonce: string, patch: Partial<PendingMarker>): Promise<void> { /* read + spread + write */ }
       async function deleteMarker(nonce: string): Promise<void> { /* unlinkSync best-effort */ }
       async function existingMarkers(): Promise<string[]> { /* fs.readdir state/.pending/ filter .json */ }

       async function sshExec(remoteCmd: string, signal?: AbortSignal): Promise<{ exit: number; stdout: string; stderr: string }> { /* Bun.spawn ssh ... */ }
       async function sshWriteFile(remotePath: string, content: string, signal?: AbortSignal): Promise<{ exit: number; stderr: string }> {
         // Bun.spawn(['ssh', ...]) + stdin pipe로 cat > <path>
         const proc = Bun.spawn(['ssh', '-o', 'ConnectTimeout=10', '-o', 'BatchMode=yes', 'gon@192.168.0.5', `cat > ${remotePath}`], { stdin: 'pipe', stdout: 'pipe', stderr: 'pipe', signal })
         proc.stdin.write(content)
         proc.stdin.end()
         const exit = await proc.exited
         return { exit, stderr: await new Response(proc.stderr).text() }
       }

       export async function applyPatch(args, ctx): Promise<ApplyPatchResult> {
         const { proposal, decision, userPrompt, remoteComposePath } = args
         const { nonce, stack, command, service, fileEdit } = proposal

         // edited decision이면 newContent override
         const effectiveFileEdit = decision.type === "edited" ? { path: proposal.fileEdit?.path ?? `state/compose/${stack}.yml`, newContent: decision.newContent, previousContent: proposal.fileEdit?.previousContent ?? "" } : proposal.fileEdit

         // D-C4: marker existence reject
         if ((await existingMarkers()).length > 0) {
           throw new Error("503: unfinished applyPatch detected (marker exists). Boot drift event 절차에 따라 복구 후 재시도.")
         }

         const t0 = Date.now()
         const sha_before = (await Bun.spawn(["git", "rev-parse", "HEAD"], { cwd: "state", stdout: "pipe" })).stdout
         const sha_before_str = await new Response(sha_before).text().then(s => s.trim())

         const remoteDir = remoteComposePath.replace(/\/[^/]+$/, "")  // 마지막 파일명 제거 → 디렉토리

         const marker: PendingMarker = {
           nonce, stack, command, service, fileEdit: effectiveFileEdit ?? undefined, reasoning: proposal.reasoning, user_prompt: userPrompt,
           sha_before: sha_before_str, ts_created: new Date().toISOString(),
           docker_started_at: null, docker_finished_at: null, exit_code: null,
         }

         // (i)
         await writeMarker(marker)

         try {
           // (ii)
           if (effectiveFileEdit) {
             await Bun.write(effectiveFileEdit.path, effectiveFileEdit.newContent)
           }

           // (iii) — fileEdit 있을 때만
           if (effectiveFileEdit) {
             const { exit, stderr } = await sshWriteFile(remoteComposePath, effectiveFileEdit.newContent, ctx.signal)
             if (exit !== 0) {
               // state 미러 revert
               await Bun.write(effectiveFileEdit.path, effectiveFileEdit.previousContent)
               await deleteMarker(nonce)
               throw new Error(`원격 파일 동기화 실패 (exit ${exit}): ${stderr.slice(0, 200)}`)
             }
           }

           // (iv)
           await updateMarker(nonce, { docker_started_at: new Date().toISOString() })
           const dockerCmd = command === "compose ps" ? "compose ps" : `${command} ${service}`
           const { exit: dockerExit, stderr: dockerStderr } = await sshExec(`cd ${remoteDir} && docker ${dockerCmd}`, ctx.signal)
           await updateMarker(nonce, { docker_finished_at: new Date().toISOString(), exit_code: dockerExit })

           if (dockerExit !== 0) {
             // (iv) fail — Open Q 3 옵션 A: 원격 파일 + state 미러 revert, git commit 안 함
             if (effectiveFileEdit) {
               await sshWriteFile(remoteComposePath, effectiveFileEdit.previousContent, ctx.signal)
               await Bun.write(effectiveFileEdit.path, effectiveFileEdit.previousContent)
             }
             await deleteMarker(nonce)
             return { outcome: { type: "rolled-back", reason: `docker exit ${dockerExit}: ${dockerStderr.slice(0, 200)}`, nonce, stack } }
           }

           // (v)
           const fileEditSummary = effectiveFileEdit ? `hunks=${(proposal.diff.match(/^@@/gm) ?? []).length} files=1` : undefined
           const action = { stack, command, service, userPrompt, reasoning: proposal.reasoning, nonce, fileEditSummary, diffSummaryRaw: undefined }
           const message = (await import("../state/commit")).buildMessage(action)

           try {
             // git add 먼저 (state 미러 + services.yaml 변경 stage)
             const addProc = Bun.spawn(["git", "add", "-A"], { cwd: "state", stdout: "pipe", stderr: "pipe" })
             const addExit = await addProc.exited
             if (addExit !== 0) throw new Error("git add 실패")

             await (await import("../state/commit")).commitWithMessage(message, nonce)

             // (vi)
             await deleteMarker(nonce)
             const sha_proc = Bun.spawn(["git", "rev-parse", "HEAD"], { cwd: "state", stdout: "pipe" })
             await sha_proc.exited
             const sha_after = (await new Response(sha_proc.stdout).text()).trim()

             return { outcome: { type: "applied", sha_after, duration_ms: Date.now() - t0, nonce, stack, command, service } }
           } catch (gitErr) {
             // (v) fail — D-B3 신규 분기
             const reverseMap: Record<string, string | null> = {
               "compose up -d": "compose down",
               "compose down": "compose up -d",
               "compose start": "compose stop",
               "compose stop": "compose start",
               "compose restart": "compose restart",
               "compose logs --tail=200": null,
               "compose ps": null,
             }
             const reverseCmd = reverseMap[command]
             let rollbackOk = false
             let rollbackCommand: string | undefined
             if (reverseCmd) {
               rollbackCommand = `${reverseCmd} ${service}`
               const rb = await sshExec(`cd ${remoteDir} && docker ${rollbackCommand}`, ctx.signal)
               rollbackOk = rb.exit === 0
             }
             // 원격 + state 파일 revert
             if (effectiveFileEdit) {
               await sshWriteFile(remoteComposePath, effectiveFileEdit.previousContent, ctx.signal)
               await Bun.write(effectiveFileEdit.path, effectiveFileEdit.previousContent)
             }
             await deleteMarker(nonce)
             return { outcome: { type: "rolled-back", reason: `git commit failed: ${(gitErr as Error).message.slice(0, 200)}`, rollback_command: rollbackCommand, rollback_ok: rollbackOk, nonce, stack } }
           }
         } catch (e) {
           // 예외가 (i)~(iii)에서 발생 시 marker delete (이미 deleted면 silent)
           await deleteMarker(nonce)
           throw e
         }
       }
