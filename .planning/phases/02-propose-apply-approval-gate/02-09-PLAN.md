    ### 3. 5-key keydown listener (window scope, form-aware)

    ```javascript
    window.addEventListener("keydown", function (ev) {
      var card = ev.target && ev.target.closest && ev.target.closest(".approval-card");
      if (!card) return;
      // textarea 안에서 입력 중이면 5-key 무시 (textarea에 normal text 입력 허용)
      if (ev.target.tagName === "TEXTAREA" || ev.target.tagName === "INPUT") return;

      var nonce = card.dataset.nonce;
      var key = ev.key.toLowerCase();
      var allowed = ["y", "n", "e", "d", "a"];
      if (allowed.indexOf(key) < 0) return;
      ev.preventDefault();

      if (key === "d") {
        var pre = card.querySelector("pre.diff");
        if (pre) pre.classList.toggle("hidden");
        return;
      }
      if (key === "e") {
        var ta = card.querySelector("textarea.edit");
        if (ta) {
          ta.classList.add("visible");
          ta.focus();
        }
        return;
      }

      // y / n / a → POST
      var body = new URLSearchParams();
      body.set("key", key);
      if (key === "y") {
        // edit textarea가 visible이면 newContent로 'edited' decision 강제
        var ta = card.querySelector("textarea.edit");
        if (ta && ta.classList.contains("visible") && ta.value.length > 0) {
          body.set("key", "e");
          body.set("newContent", ta.value);
        }
      }
      fetch("/approval/" + encodeURIComponent(nonce), {
        method: "POST",
        body: body,
        headers: { "x-session-id": (window.__sessionId || "default") },
      }).then(function (r) {
        if (!r.ok) {
          // 409 envelope — error 카드 추가
          r.json().then(function (e) {
            var err = document.createElement("div");
            err.className = "error-card";
            err.textContent = "✗ approval 실패: " + (e.problem || "");
            outputEl.appendChild(err);
          });
          return;
        }
        // 성공 — applied 또는 rolled-back SSE event가 곧 도착
        // 카드 자체에 'submitted' 표식
        card.classList.add("submitted");
        var status = document.createElement("div");
        status.className = "submitted-status";
        status.textContent = "✓ 결정 전송: " + key + " (서버 응답 대기...)";
        card.appendChild(status);
      }).catch(function (e) {
        var err = document.createElement("div");
        err.className = "error-card";
        err.textContent = "✗ 전송 실패: " + (e.message || e);
        outputEl.appendChild(err);
      });
    });
    ```
