    구현 핵심:
    ```typescript
    // agent/loop.ts (확장)
    import { proposePatch } from "../tools/proposePatch"
    import { applyPatch } from "../tools/applyPatch"
    import { expirePending } from "../approval/store"

    // remoteComposePath 매핑 — 02-04 SUMMARY에 lock된 5 경로
    const REMOTE_COMPOSE_PATH: Record<string, string> = {
      "news":       "/home/gon/projects/gon/news-sentiment-prod/docker-compose.yml",
      "ais":        "/home/gon/projects/gon/ai-afterschool-services/docker-compose.prod.yml",
      "n8n":        "/home/gon/projects/gon/n8n/docker-compose.yml",
      "open-webui": "/home/gon/projects/gon/open-webui/docker-compose.yml",
      "krdn-fx":    "/home/gon/projects/gon/krdn-fx/docker-compose.yml",
    }
    // ⚠ 02-04에서 정확한 경로로 갱신될 것 — 본 plan 작성 시점은 placeholder

    async function dispatch(name, input, turnId, sessionId, userPrompt, emit, ac) {
      switch (name) {
        // ... 기존 3 case
        case "proposePatch": {
          // _envelope.run wrap (audit row + 30s timeout)
          const wrapped = await run("proposePatch", async (signal) => {
            return await proposePatch(input as ProposePatchArgs, { sessionId, userPrompt })
          }, 30_000, turnId, input)

          if (wrapped && "problem" in wrapped) return wrapped  // ToolError envelope
          const { result, decision } = wrapped as { result: ProposePatchResult; decision: Promise<ApprovalDecision> }

          // emit approval-required
          emit({ type: "approval-required", nonce: result.nonce, stack: result.stack, command: result.command, service: result.service, diff: result.diff, reasoning: result.reasoning, expiresAt: result.expiresAt })

          // keep-alive interval (advisor lock)
          const keepAliveInterval = setInterval(() => {
            emit({ type: "text-delta", delta: "" })
          }, 30_000)

          // abort wire
          const abortHandler = () => { clearInterval(keepAliveInterval); expirePending(sessionId) }
          ac?.signal.addEventListener("abort", abortHandler, { once: true })

          let resolved: ApprovalDecision
          try {
            resolved = await decision
          } finally {
            clearInterval(keepAliveInterval)
            ac?.signal.removeEventListener("abort", abortHandler)
          }

          if (resolved.type === "approved" || resolved.type === "edited") {
            // applyPatch internal call
            const apply = await run("applyPatch", async (signal) => {
              return await applyPatch({
                proposal: result,
                decision: resolved,
                userPrompt,
                remoteComposePath: REMOTE_COMPOSE_PATH[result.stack],
              }, { signal })
            }, 60_000, turnId, { proposal: result, decision: resolved })

            if (apply && "problem" in apply) {
              // _envelope.run에서 throw 잡힘 — emit + ToolError 반환
              emit({ type: "rolled-back", nonce: result.nonce, stack: result.stack, reason: (apply as any).problem })
              return apply
            }

            const outcome = (apply as ApplyPatchResult).outcome
            if (outcome.type === "applied") {
              emit({ type: "applied", nonce: outcome.nonce, stack: outcome.stack, command: outcome.command, service: outcome.service, sha_after: outcome.sha_after, duration_ms: outcome.duration_ms })
              return { ok: true, ...outcome }
            } else {
              emit({ type: "rolled-back", nonce: outcome.nonce, stack: outcome.stack, reason: outcome.reason, rollback_command: outcome.rollback_command, rollback_ok: outcome.rollback_ok })
              return { problem: `applyPatch rolled-back: ${outcome.reason}`, cause: "docker exec or git commit failed", fix: "수동 점검 후 재시도 (rollback_command 결과 확인)", retryable: false }
            }
          }
          // rejected / aborted / expired
          const reasonMap = {
            "rejected": { problem: "approval rejected", cause: "사용자 'n' 키", fix: "다른 접근(read tool로 상태 재확인)을 시도", retryable: false },
            "aborted":  { problem: "approval aborted",  cause: "사용자 'a' 키", fix: "문맥 재구성 후 다시 제안", retryable: false },
            "expired":  { problem: "approval expired (2 min)", cause: "운영자 미응답", fix: "다시 proposePatch 호출 (원하면)", retryable: true },
          }
          return reasonMap[resolved.type as "rejected" | "aborted" | "expired"]
        }
        default: /* 기존 */
      }
    }
    ```
