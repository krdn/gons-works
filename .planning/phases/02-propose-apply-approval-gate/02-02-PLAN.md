  nonce: string                          // crypto.randomUUID() per proposePatch
  diff: string                            // jsdiff createPatch 결과 또는 "(no file change ...)"
  stack: Stack
  command: ComposeCommand
  service: string                         // <svc> (ps 제외 시 의미 있음, ps는 빈 문자열 OK)

  expiresAt: number                       // Date.now() + 120_000
  consumed: boolean                       // atomic set-before-resolve (PITFALL #3)
}

export type ApprovalDecision =
  | { type: "approved" }
  | { type: "rejected" }
  | { type: "edited"; newContent: string }   // 'e' 키: textarea 로컬 편집 결과 (Open Q 1 lock — LLM 재제안은 v2)
  | { type: "aborted" }
  | { type: "expired" }

// LOOP에서 호출: proposePatch tool이 PendingApproval 등록 + Promise 반환 받음
export function setPending(sessionId: string, payload: Omit<PendingApproval, "consumed">): Promise<ApprovalDecision>

// LOOP는 setPending이 반환한 Promise를 직접 await (advisor Open Q 2 lock).
// waitForApproval은 export하지 않음 — sessionId만으로 Promise 재 attach가 불가능하기 때문.

// server.ts /approval/:id POST 핸들러에서 호출
// - nonce mismatch / consumed / expired 모두 throw
// - 통과 시 atomic consumed=true → resolve(decision)
export function consumeApproval(sessionId: string, clientNonce: string, decision: ApprovalDecision): void

// 외부 abort (loop AbortController가 trigger) — resolve(expired) + delete
export function expirePending(sessionId: string): void

// test only
export function _resetForTest(): void

    2. **setPending(sessionId, payload):**
       - 같은 sessionId 이미 존재 → `throw new Error("Pending approval already exists for session: " + sessionId)`
       - `consumed: false` 보강 + Promise 생성
       - `setTimeout(() => { ... resolve({type:"expired"}); pending.delete(sessionId) }, payload.expiresAt - Date.now())`로 자동 expire 등록 (timeoutHandle 보관)
       - Map에 set
       - Promise 반환

    3. **consumeApproval(sessionId, clientNonce, decision):** PITFALLS.md `Pitfall 3` lines 117-141 prevention 코드 verbatim:
       ```typescript
       const a = pending.get(sessionId)
       if (!a) throw new Error("No pending approval")
       if (a.nonce !== clientNonce) throw new Error("Stale approval — nonce mismatch. Re-read the diff before approving.")
       if (a.consumed) throw new Error("Already consumed (double-press guard)")
       if (Date.now() > a.expiresAt) {
         a.resolve({ type: "expired" })
         clearTimeout(a.timeoutHandle)
         pending.delete(sessionId)
         throw new Error("Approval expired (2 min). Re-propose.")
       }
       a.consumed = true               // BEFORE resolve
       a.resolve(decision)
       clearTimeout(a.timeoutHandle)
       pending.delete(sessionId)
       ```

    4. **expirePending(sessionId):**
       - pending get → 있으면 resolve({type:"expired"}) + clearTimeout + delete
       - 없으면 silent no-op (loop AbortController가 race로 두 번 호출할 수 있음)

    5. **_resetForTest():** `for (const a of pending.values()) clearTimeout(a.timeoutHandle); pending.clear()`
