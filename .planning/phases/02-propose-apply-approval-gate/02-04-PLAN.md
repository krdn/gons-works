    import { join } from "node:path"

    // Option A hotfix 2026-05-07 — 4 stack scope (open-webui 제외, v2 backlog)
    // 경로는 executor가 `docker inspect <container> | jq -r '.[0].Config.Labels["com.docker.compose.project.config_files"]'`로 검증

      "news":    "/home/gon/deploy/news-sentiment-prod/docker-compose.yml",
      "ais":     "/home/gon/actions-runner/_work/ai-signalcraft/ai-signalcraft/docker/docker-compose.prod.yml",
      "n8n":     "/home/gon/docker-n8n/docker-compose.yml",
      "krdn-fx": "/home/gon/projects/krdn-fx/docker-compose.yml",
    }

    async function sshCat(remotePath: string): Promise<string> {
      const proc = Bun.spawn(
        [
          "ssh",
          "-o", "ConnectTimeout=10",
          "-o", "ServerAliveInterval=5",
          "-o", "ServerAliveCountMax=2",
          "-o", "BatchMode=yes",
          "gon@192.168.0.5",
          "cat", remotePath,
        ],
        { stdout: "pipe", stderr: "pipe" },
      )
      const exit = await proc.exited
      if (exit !== 0) {
        const err = await new Response(proc.stderr).text()
        throw new Error(`SSH cat 실패 ${remotePath}: ${err.slice(0, 200)}`)
      }
      return await new Response(proc.stdout).text()
    }

    async function gitCmd(args: string[]): Promise<void> {
      const proc = Bun.spawn(["git", ...args], { cwd: "state", stdout: "pipe", stderr: "pipe" })
      const exit = await proc.exited
      if (exit !== 0) {
        const err = await new Response(proc.stderr).text()
        throw new Error(`git ${args.join(" ")} 실패: ${err.slice(0, 200)}`)
      }
    }

    async function main() {
      const composeDir = "state/compose"
      if (!existsSync(composeDir)) mkdirSync(composeDir, { recursive: true })

      for (const [stack, remotePath] of Object.entries(REMOTE_COMPOSE_PATHS)) {
        console.log(`[init] ${stack} ← ${remotePath}`)
        const content = await sshCat(remotePath)
        await Bun.write(join(composeDir, `${stack}.yml`), content)
      }

      // git status 확인 → 변경 있을 때만 commit
      const status = Bun.spawn(["git", "status", "--porcelain"], { cwd: "state", stdout: "pipe" })
      await status.exited
      const out = (await new Response(status.stdout).text()).trim()
      if (out.length === 0) {
        console.log("[init] no changes, skipping commit")
        return
      }

      await gitCmd(["add", "compose/"])
      await gitCmd(["commit", "-m", "feat(02): bootstrap state/compose mirror (5 stacks via SSH cat)"])
      console.log("[init] OK — 5 mirrors committed.")
    }

    main().catch((e) => { console.error(e); process.exit(1) })
    ```

    // tests/setup/docker-context.ts
    export async function switchToLocalContext(): Promise<{ originalContext: string; testStackUp: boolean }> {
      const originalContext = process.env.DOCKER_CONTEXT ?? ""
      process.env.DOCKER_CONTEXT = "default"
      const proc = Bun.spawn(
        ["docker", "compose", "-f", "tests/fixtures/test-compose.yml", "up", "-d"],
        { stdout: "pipe", stderr: "pipe" },
      )
      const exit = await proc.exited
      if (exit !== 0) {
        const err = await new Response(proc.stderr).text()
        throw new Error(`docker compose up 실패: ${err.slice(0, 200)}`)
      }
      return { originalContext, testStackUp: true }
    }

    export async function restoreContext(saved: { originalContext: string; testStackUp: boolean }): Promise<void> {
      if (saved.testStackUp) {
        const proc = Bun.spawn(
          ["docker", "compose", "-f", "tests/fixtures/test-compose.yml", "down", "--volumes"],
          { stdout: "pipe", stderr: "pipe" },
        )
        await proc.exited  // exit 무시 — cleanup best-effort
      }
      if (saved.originalContext === "") {
        delete process.env.DOCKER_CONTEXT
      } else {
        process.env.DOCKER_CONTEXT = saved.originalContext
      }
    }
    ```
