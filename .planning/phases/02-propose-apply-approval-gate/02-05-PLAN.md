```typescript
import { z } from "zod"

export const ProposePatchInput = z.object({
  stack: z.enum(["news", "ais", "n8n", "krdn-fx"]),  // D-A1 + Option A 4-stack scope (open-webui 제외, v2 backlog — plain docker run)
  command: z.union([                                                 // D-B1 7-union literal
    z.literal("compose up -d"),
    z.literal("compose down"),
    z.literal("compose restart"),
    z.literal("compose start"),
    z.literal("compose stop"),
    z.literal("compose logs --tail=200"),
    z.literal("compose ps"),
  ]),
  service: z.string().min(1).optional(),  // 'compose ps'는 svc 없이 OK; 나머지는 필수 (refine으로 강제)
  fileEdit: z.object({
    path: z.string(),                     // strict: state/compose/{stack}.yml | state/services.yaml — 검증은 refine
    newContent: z.string(),
  }).optional(),
  reasoning: z.string().min(1).max(500),  // D-D2 필수
}).superRefine((val, ctx) => {
  if (val.command !== "compose ps" && (!val.service || val.service.length === 0)) {
    ctx.addIssue({ code: z.ZodIssueCode.custom, message: "service 필드는 'compose ps' 외 필수" })
  }
  if (val.fileEdit) {
    const allowed = [`state/compose/${val.stack}.yml`, "state/services.yaml"]
    if (!allowed.includes(val.fileEdit.path)) {
      ctx.addIssue({ code: z.ZodIssueCode.custom, message: `fileEdit.path는 ${allowed.join(" 또는 ")}만 허용` })
    }
  }
})

  // SSE approval-required event payload용

  stack: string
  command: string
  service: string
```

    ```typescript
    import { createPatch } from "diff"
    import { z } from "zod"
    import { run } from "./_envelope"
    import { setPending, type ApprovalDecision } from "../approval/store"

    export const ProposePatchInput = z.object({ ... })  // 위 interfaces 블록 그대로
    export type ProposePatchArgs = z.infer<typeof ProposePatchInput>

    export interface ProposePatchResult { ... }

    export async function proposePatch(
      args: ProposePatchArgs,
      ctx: { sessionId: string; userPrompt: string },
    ): Promise<{ result: ProposePatchResult; decision: Promise<ApprovalDecision> }> {
      // _envelope.run wrap은 caller(loop) 책임 — 본 함수는 순수 service.
      // 단, run() 패턴 일관성 유지 위해 LOOP에서 run("proposePatch", () => proposePatch(...)) 형태로 호출.

      const parsed = ProposePatchInput.parse(args)
      let diff: string
      let previousContent: string | undefined
      if (parsed.fileEdit) {
        try {
          previousContent = await Bun.file(parsed.fileEdit.path).text()
        } catch {
          previousContent = ""  // 파일 없는 경우 — diff는 0→new 전체 추가
        }
        diff = createPatch(parsed.fileEdit.path, previousContent, parsed.fileEdit.newContent, "current", "proposed")
      } else {
        diff = `(no file change — command only: ${parsed.command}${parsed.service ? " " + parsed.service : ""})`
      }

      const nonce = crypto.randomUUID()
      const expiresAt = Date.now() + 120_000  // 2분 (PITFALL #3)

      const decision = setPending(ctx.sessionId, {
        nonce,
        diff,
        stack: parsed.stack,
        command: parsed.command,
        service: parsed.service ?? "",
        fileEdit: parsed.fileEdit,
        reasoning: parsed.reasoning,
        user_prompt: ctx.userPrompt,
        expiresAt,
      })

      const result: ProposePatchResult = {
        nonce,
        stack: parsed.stack,
        command: parsed.command,
        service: parsed.service ?? "",
        diff,
        reasoning: parsed.reasoning,
        expiresAt,
        fileEdit: parsed.fileEdit ? { ...parsed.fileEdit, previousContent: previousContent ?? "" } : undefined,
      }

      return { result, decision }
    }
    ```

    - service가 'compose ps'에서는 빈 문자열로 두지만 result에는 명시적 빈 문자열 (string, undefined 아님).
