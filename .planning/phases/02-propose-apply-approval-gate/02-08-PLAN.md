    import { consumeApproval } from "../approval/store"

    function mapKey(key: string, newContent?: string) {
      switch (key) {
        case "y": return { type: "approved" as const }
        case "n": return { type: "rejected" as const }
        case "e": return newContent ? { type: "edited" as const, newContent } : null
        case "d": return "ui-only" as const
        case "a": return { type: "aborted" as const }
        default: return null
      }
    }

    app.post("/approval/:id", async (c) => {
      const nonce = c.req.param("id")
      const sessionId = c.req.header("x-session-id") ?? "default"
      let body: Record<string, string | File>
      try {
        body = await c.req.parseBody()
      } catch {
        return c.json({ problem: "invalid body", cause: "form parse 실패", fix: "form-urlencoded body로 재전송", retryable: false }, 400)
      }
      const key = String(body.key ?? "")
      const newContent = body.newContent != null ? String(body.newContent) : undefined

      const mapped = mapKey(key, newContent)
      if (mapped === null) {
        return c.json({ problem: "invalid key", cause: `key='${key}' (또는 'e' newContent 누락)`, fix: "y/n/e/d/a 중 하나 + 'e'면 newContent 동봉", retryable: false }, 400)
      }
      if (mapped === "ui-only") {
        return c.json({ ok: true, ui_only: true })
      }
      try {
        consumeApproval(sessionId, nonce, mapped)
        return c.json({ ok: true })
      } catch (err) {
        const msg = (err as Error).message
        return c.json({ problem: msg, cause: "approval store invariant violation", fix: "diff 다시 읽고 재제안 또는 새 NL 입력", retryable: false }, 409)
      }
    })
    ```

    import { readdir, readFile } from "node:fs/promises"
    import { existsSync } from "node:fs"

    interface PendingMarker {
      nonce: string
      stack: string
      command: string
      service: string
      docker_started_at: string | null
      docker_finished_at: string | null
      exit_code: number | null
      ts_created: string
    }

    let pendingDriftBuffer: PendingMarker[] = []

    async function recoverPendingMarkers(): Promise<PendingMarker[]> {
      const dir = "state/.pending"
      if (!existsSync(dir)) return []
      const files = (await readdir(dir)).filter(f => f.endsWith(".json"))
      if (files.length === 0) return []
      const markers: PendingMarker[] = []
      for (const f of files) {
        try {
          const content = await readFile(`${dir}/${f}`, "utf-8")
          markers.push(JSON.parse(content))
        } catch {
          /* swallow — corrupt marker는 운영자 manual */
        }
      }
      return markers
    }

    function bufferPendingDrift(markers: PendingMarker[]): void {
      pendingDriftBuffer = markers
    }

    function consumeBufferedDrift(): PendingMarker[] {
      const out = pendingDriftBuffer
      pendingDriftBuffer = []
      return out
    }

    // startup() 함수에 5번째 probe 추가
    async function startup(): Promise<void> {
      // probe 1-4 (Phase 1 carry-forward)
      ...
      // probe 5: pending markers (D-C1, D-C2)
      try {
        const markers = await recoverPendingMarkers()
        if (markers.length > 0) {
          console.warn(`[APPLY-05] ${markers.length} unfinished applyPatch detected — drift event will fire on next /chat-stream`)
          for (const m of markers) {
            console.warn(`  - nonce=${m.nonce} stack=${m.stack} command=${m.command} docker_started_at=${m.docker_started_at} exit_code=${m.exit_code}`)
          }
          bufferPendingDrift(markers)
        }
      } catch (e) {
        console.warn(`[APPLY-05] recoverPendingMarkers 실패 (server start 계속): ${(e as Error).message}`)
      }
    }
    ```
