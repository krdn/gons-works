    import { $ } from "bun"
    import { loadEnv } from "../src/env"
    import { run, type ToolError } from "./_envelope"
    import type { ReadLogsArgs } from "./_index"

    // KB-04: PITFALL #6 prevention block 그대로 채택
    const SENSITIVE_PATTERNS = [
      /\b(API_KEY|SECRET|PASSWORD|TOKEN|BEARER|AUTH)\s*[:=]\s*\S+/gi,
      /sk-[a-zA-Z0-9]{20,}/g,
      /eyJ[a-zA-Z0-9_-]{20,}/g,
    ]

    const INJECTION_MARKERS = [
      /SYSTEM\s+(OVERRIDE|PROMPT|MESSAGE)\s*:/i,
      /ignore\s+prior\s+context/i,
      /<\|im_start\|>/i,
      /\[INST\]/i,
    ]

    export function sanitizeLogs(raw: string): string {
      return raw
        .split("\n")
        .filter((line) => !INJECTION_MARKERS.some((p) => p.test(line)))
        .map((line) => SENSITIVE_PATTERNS.reduce((l, p) => l.replace(p, "[REDACTED]"), line))
        .join("\n")
    }

    export async function readLogs(args: ReadLogsArgs): Promise<string | ToolError> {
      const env = loadEnv()
      const ctx = env.DOCKER_CONTEXT
      const { containerName, lines, since, until } = args
      return await run("readLogs", async (_signal) => {
        // PITFALL #8: ctx를 Bun.$ template에 명시 (process.env.DOCKER_CONTEXT 직접 금지)
        let cmd = $`docker --context ${ctx} logs --tail ${lines} ${containerName}`
        if (since && until) {
          cmd = $`docker --context ${ctx} logs --tail ${lines} --since ${since} --until ${until} ${containerName}`
        } else if (since) {
          cmd = $`docker --context ${ctx} logs --tail ${lines} --since ${since} ${containerName}`
        }
        const raw = (await cmd.text()) ?? ""
        return sanitizeLogs(raw)
      })
    }

    import { loadEnv } from "../src/env"
    import { run, type ToolError } from "./_envelope"

    export interface DockerContainer {
      name: string
      image: string
      state: string
      status: string
    }

    export async function listContainers(args: ListContainersArgs = { limit: 20 }): Promise<DockerContainer[] | ToolError> {
      const env = loadEnv()
      const ctx = env.DOCKER_CONTEXT
      const stateFilter = args.filter?.state ?? "running"
      const namePattern = args.filter?.namePattern
      const limit = args.limit ?? 20

      return await run("listContainers", async (_signal) => {
        // PITFALL #8: --context 명시. NDJSON 파싱 (draft-services-yaml.ts 패턴 재사용)
        const flag = stateFilter === "all" ? "-a" : ""
        const raw = (await $`docker --context ${ctx} ps ${flag} --format json`.text()) ?? ""
        const rows = raw.trim().split("\n").filter((l) => l.length > 0)
          .map((l) => JSON.parse(l) as { Names: string; Image: string; State: string; Status: string })

        const filtered = namePattern
          ? rows.filter((r) => r.Names.includes(namePattern))
          : rows
        const stateMatched = stateFilter === "all" ? filtered : filtered.filter((r) =>
          stateFilter === "running" ? r.State === "running" : r.State !== "running"
        )

        return stateMatched.slice(0, limit).map((r) => ({
          name: r.Names,
          image: r.Image,
          state: r.State,
          status: r.Status,
        }))
      })
    }

    // 기존 ToolError + isToolError 정의는 그대로 유지

    export async function run<T>(
      name: string,
      fn: (signal: AbortSignal) => Promise<T>,
      timeoutMs = 30_000,
      auditParentId?: bigint,   // 신규 optional — Plan 06 agent/loop.ts 진입점에서만 전달
      auditInput?: unknown,     // 신규 optional — original tool args
    ): Promise<T | ToolError> {
      const ac = new AbortController()
      const timer = setTimeout(() => ac.abort(), timeoutMs)
      const startedAt = Date.now()
      let result: T | ToolError
      let ok = false
      try {
        result = await fn(ac.signal)
        ok = !isToolError(result)
        return result
      } catch (err) {
        // PITFALL #1: 절대 throw — 항상 ToolError 반환
        if ((err as Error).name === "AbortError") {
          result = {
            problem: `tool ${name} timeout ${timeoutMs / 1000}s`,
            cause: "서버 도달·응답 지연",
            fix: "재시도 또는 SSH/docker context 상태 점검",
            retryable: true,
          }
        } else {
          result = {
            problem: `tool ${name} 실패`,
            cause: ((err as Error).message ?? String(err)).slice(0, 200),
            fix: "로그 확인 후 재시도",
            retryable: true,
          }
        }
        return result
      } finally {
        clearTimeout(timer)
        // AUDIT-01 / D-14.4: parentId가 전달되면 logTool 호출. 없으면 skip (test simplicity).
        if (auditParentId !== undefined) {
          const summary = isToolError(result!)
            ? `${(result as ToolError).problem}: ${(result as ToolError).cause}`

          try {
            logTool(auditParentId, name, auditInput ?? null, summary, ok, Date.now() - startedAt)
          } catch {
            // audit 실패가 tool 실행 결과에 영향 주지 않도록 swallow
          }
        }
      }
    }
