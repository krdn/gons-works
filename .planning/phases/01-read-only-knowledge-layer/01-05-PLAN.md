    ```typescript
    import type { Stack, ServicesYaml } from "./schema"

    export interface Chunk {
      id: string
      text: string
      metadata: {
        stack: string
        field: "purpose" | "depends_on" | "volumes" | "normal_log_pattern" | "key_log_locations"
      }
    }

    type FieldTemplate = (stackName: string, val: unknown) => string

    export const FIELD_TEMPLATES: Record<Chunk["metadata"]["field"], FieldTemplate> = {
      purpose: (s, v) => `${s} stack: ${(v as string) || "정의 없음"}.`,
      depends_on: (s, v) => {
        const arr = v as string[]
        if (!arr || arr.length === 0) return `${s} stack의 의존성: 정의 없음.`
        return `${s} stack은 ${arr.join(", ")}에 의존한다.`
      },
      volumes: (s, v) => {
        const arr = v as string[]
        if (!arr || arr.length === 0) return `${s} stack의 볼륨: 정의 없음.`
        return `${s} stack의 볼륨: ${arr.join(", ")}.`
      },
      normal_log_pattern: (s, v) => {
        const pattern = v as string
        if (!pattern) return `${s} stack의 normal_log_pattern: 정의 없음.`
        return `${s} stack의 normal_log_pattern: "${pattern}". 이 패턴이 안 보이면 worker 멈춤이나 에러 의심.`
      },
      key_log_locations: (s, v) => {
        const arr = v as string[]
        if (!arr || arr.length === 0) return `${s} 디버깅 시 확인할 컨테이너: 정의 없음.`
        return `${s} 디버깅 시 확인할 컨테이너: ${arr.join(", ")}.`
      },
    }

    export function chunkStack(stackName: string, stack: Stack): Chunk[] {
      return (Object.keys(FIELD_TEMPLATES) as Array<Chunk["metadata"]["field"]>).map((field) => ({
        id: `${stackName}:${field}`,
        text: FIELD_TEMPLATES[field](stackName, stack[field as keyof Stack]),
        metadata: { stack: stackName, field },
      }))
    }

    export function chunkServicesYaml(yaml: ServicesYaml): Chunk[] {
      return Object.entries(yaml.stacks).flatMap(([stackName, stack]) =>
        chunkStack(stackName, stack as Stack)
      )
    }
    ```

    ```typescript
    import { Database } from "bun:sqlite"
    import { VoyageAIClient } from "voyageai"
    import { createHash } from "node:crypto"
    import { mkdirSync, existsSync } from "node:fs"
    import { dirname } from "node:path"
    import { load } from "js-yaml"
    import { ServicesYamlSchema } from "./schema"
    import { chunkServicesYaml, type Chunk } from "./chunker"
    import { loadEnv } from "../src/env"

    const KB_DB_PATH = "./data/copilot.db"
    const VOYAGE_MODEL = "voyage-4-lite"   // DESIGN.md correction #1 + STACK.md lock
    let _db: Database | null = null

    function getDb(): Database {
      if (_db) return _db
      const dir = dirname(KB_DB_PATH)
      if (!existsSync(dir)) mkdirSync(dir, { recursive: true })
      // PITFALL #10: strict + safeIntegers
      _db = new Database(KB_DB_PATH, { strict: true, safeIntegers: true, create: true })
      _db.run("PRAGMA journal_mode = WAL")
      _db.run("PRAGMA synchronous = NORMAL")
      _db.run(`CREATE TABLE IF NOT EXISTS kb_meta (key TEXT PRIMARY KEY, value TEXT NOT NULL)`)
      _db.run(`CREATE TABLE IF NOT EXISTS kb_chunks (
        id TEXT PRIMARY KEY,
        text TEXT NOT NULL,
        stack TEXT NOT NULL,
        field TEXT NOT NULL,
        embedding TEXT NOT NULL  -- JSON-encoded float32 array (1024 dim)
      )`)
      return _db
    }

    export function cosineSimilarity(a: number[], b: number[]): number {
      if (a.length !== b.length) throw new Error("벡터 차원 불일치")
      let dot = 0, normA = 0, normB = 0
      for (let i = 0; i < a.length; i++) {
        dot += a[i]! * b[i]!
        normA += a[i]! * a[i]!
        normB += b[i]! * b[i]!
      }
      if (normA === 0 || normB === 0) return 0
      return dot / (Math.sqrt(normA) * Math.sqrt(normB))
    }

    async function embedDocuments(client: VoyageAIClient, texts: string[]): Promise<number[][]> {
      const resp = await client.embed({ input: texts, model: VOYAGE_MODEL, inputType: "document" })
      return (resp.data ?? []).map((d) => d.embedding ?? [])
    }

    async function embedQuery(client: VoyageAIClient, query: string): Promise<number[]> {
      const resp = await client.embed({ input: [query], model: VOYAGE_MODEL, inputType: "query" })
      return resp.data?.[0]?.embedding ?? []
    }

    // D-13.3: hash 기반 lazy indexing
    export async function ensureIndexed(force = false): Promise<{ indexed: boolean; chunkCount: number }> {
      const env = loadEnv()
      const db = getDb()
      const yaml = await Bun.file("state/services.yaml").text()
      const hash = createHash("sha256").update(yaml).digest("hex")

      if (!force) {
        const stored = db.prepare("SELECT value FROM kb_meta WHERE key = $key").get({ $key: "yaml_hash" }) as { value: string } | null
        if (stored?.value === hash) {
          const count = (db.query("SELECT COUNT(*) AS c FROM kb_chunks").get() as { c: bigint }).c
          return { indexed: false, chunkCount: Number(count) }
        }
      }

      // 재인덱싱 path
      const parsed = ServicesYamlSchema.parse(load(yaml))
      const chunks = chunkServicesYaml(parsed)
      const client = new VoyageAIClient({ apiKey: env.VOYAGE_API_KEY })
      const vectors = await embedDocuments(client, chunks.map((c) => c.text))

      db.run("DELETE FROM kb_chunks")
      const insert = db.prepare(
        "INSERT INTO kb_chunks (id, text, stack, field, embedding) VALUES ($id, $text, $stack, $field, $emb)"
      )
      for (let i = 0; i < chunks.length; i++) {
        const c = chunks[i]!
        const v = vectors[i] ?? []
        insert.run({
          $id: c.id,
          $text: c.text,
          $stack: c.metadata.stack,
          $field: c.metadata.field,
          $emb: JSON.stringify(v),
        })
      }
      db.prepare("INSERT OR REPLACE INTO kb_meta (key, value) VALUES ($key, $value)").run({ $key: "yaml_hash", $value: hash })
      return { indexed: true, chunkCount: chunks.length }
    }

    export interface KbHit {
      id: string
      text: string
      stack: string
      field: string
      similarity: number
    }

    // KB-02: top-k=5 cosine retrieval
    export async function queryTopK(query: string, k = 5): Promise<KbHit[]> {
      const env = loadEnv()
      const db = getDb()
      const client = new VoyageAIClient({ apiKey: env.VOYAGE_API_KEY })
      const qVec = await embedQuery(client, query)
      const rows = db.query("SELECT id, text, stack, field, embedding FROM kb_chunks").all() as Array<{ id: string; text: string; stack: string; field: string; embedding: string }>
      const scored: KbHit[] = rows.map((r) => ({
        id: r.id,
        text: r.text,
        stack: r.stack,
        field: r.field,
        similarity: cosineSimilarity(qVec, JSON.parse(r.embedding) as number[]),
      }))
      scored.sort((a, b) => b.similarity - a.similarity)
      return scored.slice(0, k)
    }

    export function _resetForTest(): void {
      if (_db) { _db.close(); _db = null }
    }
    ```

    ```typescript
      stale: string[]      // yaml-only stack — 라이브 인스턴스 0
      drifted: string[]    // Phase 1: 빈 배열 (v2 deferred)
      checkedAt: string    // ISO timestamp
    }

    interface CacheEntry { report: StalenessReport; ts: number }
    const CACHE_TTL_MS = 30_000
    const cache = new Map<string, CacheEntry>()

    export function getStalenessCache(): Map<string, CacheEntry> {
      return cache
    }

    export async function staleCheck(env: Env): Promise<StalenessReport> {
      const yaml = await Bun.file("state/services.yaml").text()
      const cacheKey = `${env.DOCKER_CONTEXT}:${yaml.length}`
      const cached = cache.get(cacheKey)
      if (cached && Date.now() - cached.ts < CACHE_TTL_MS) return cached.report

      // PITFALL #8: env.DOCKER_CONTEXT 명시
      const raw = (await $`docker --context ${env.DOCKER_CONTEXT} ps -a --format json`.text()) ?? ""
      const containers = raw.trim().split("\n").filter(Boolean).map((l) => JSON.parse(l) as { Names: string; State: string })
      const parsed = ServicesYamlSchema.parse(load(yaml))

      const yamlStacks = new Set(Object.keys(parsed.stacks))
      const dockerStackHits = new Set<string>()
      const unknownContainers: string[] = []

      for (const c of containers) {
        const stack = classifyStack(c.Names)
        if (stack && yamlStacks.has(stack)) {
          // 라이브 매칭 — running 만 stack hit으로 카운트 (yaml stale 판단)
          if (c.State === "running") dockerStackHits.add(stack)
        } else if (!stack && c.State === "running") {
          // 5 stack 미매칭 + running → unknown
          unknownContainers.push(c.Names)
        }
      }

      const staleStackKeys = [...yamlStacks].filter((k) => !dockerStackHits.has(k))

      const report: StalenessReport = {
        unknown: unknownContainers,
        stale: staleStackKeys,
        drifted: [],   // Phase 1 lock
        checkedAt: new Date().toISOString(),
      }
      cache.set(cacheKey, { report, ts: Date.now() })
      return report
    }
    ```
