    ```typescript
    import { Hono } from "hono"
    import { streamSSE } from "hono/streaming"
    import { existsSync, readFileSync } from "node:fs"
    import { resolve as pathResolve } from "node:path"
    import { loadEnv } from "./env"
    import { ensureDockerContext } from "./docker-context-check"
    import { ensureIndexed, queryTopK } from "../kb/index"
    import { staleCheck, type StalenessReport } from "../kb/stale-check"
    import { iterate } from "../agent/loop"
    import { toSSEFrame, createChunkWatchdog, type SseEvent } from "../agent/sse"
    import { formatRagContext } from "../agent/system-prompt"

    const env = loadEnv()
    const app = new Hono()

    // ===== Static file serving (htmx + sse extension) =====
    function serveStatic(path: string): Response | null {
      // STACK.md lock: htmx@2.0.10, htmx-ext-sse@2.2.4 — node_modules에서 serve
      const candidates: Record<string, string> = {
        "htmx.min.js": "node_modules/htmx.org/dist/htmx.min.js",
        "sse.js": "node_modules/htmx-ext-sse/sse.js",
      }
      const fpath = candidates[path]
      if (!fpath || !existsSync(fpath)) return null
      const body = readFileSync(fpath)
      return new Response(body, { headers: { "Content-Type": "application/javascript" } })
    }

    app.get("/static/:file", (c) => {
      const file = c.req.param("file")
      const resp = serveStatic(file)
      return resp ?? c.text(`static not found: ${file}`, 404)
    })

    // ===== root index.html =====
    app.get("/", (c) => {
      const html = readFileSync("public/index.html", "utf-8")
      return c.html(html)
    })

    // ===== POST /chat — htmx 폼 submit endpoint (response: 빈 200, SSE는 sse-connect로 별도 연결) =====
    app.post("/chat", async (c) => {
      // htmx hx-post는 일반 응답을 받지만, SSE 스트림은 GET /chat-stream으로 별도 연결.
      // POST /chat은 prompt를 in-memory queue에 저장 후 /chat-stream이 consume.
      const body = await c.req.parseBody()
      const prompt = String(body.prompt ?? "").trim()
      if (!prompt) return c.json({ error: "prompt 비어있음" }, 400)
      // sessionId는 Cookie 또는 query string (Plan 08 frontend가 결정).
      // Phase 1: 단순화를 위해 prompt 자체를 GET /chat-stream?prompt=... 으로 redirect.
      // 운영자 1명, 단일 페이지 → race condition 무시 가능.
      return c.html(`<div hx-ext="sse" sse-connect="/chat-stream?prompt=${encodeURIComponent(prompt)}" hx-swap="none"></div>`)
      // hx-swap=none + sse-connect = 위 div가 #output 이외 영역에 삽입되며 SSE 시작
    })

    // ===== GET /chat-stream — SSE handler (READ-04 + READ-05 + UI-04 + LOOP-06) =====
    app.get("/chat-stream", (c) => {
      const prompt = c.req.query("prompt") ?? ""
      const sessionId = c.req.header("x-session-id") ?? "default"

      return streamSSE(c, async (stream) => {
        // PITFALL #4: AbortError catch (server crash 방지)
        // LOOP-06 / D-15.3: 60s no-chunk watchdog — SSE leak 방지.
        // emit()마다 reset, final/error마다 cancel.
        let watchdogFired = false
        const watchdog = createChunkWatchdog((timeoutEvent) => {
          watchdogFired = true
          // watchdog 자체가 에러 envelope을 만들어 emit. emit()이 throw해도 stream close 진행.
          void stream.writeSSE(toSSEFrame(timeoutEvent)).catch(() => { /* stream already closed */ })
        })

        const emit = async (ev: SseEvent) => {
          watchdog.reset()                                  // LOOP-06: 매 chunk마다 timer 갱신
          await stream.writeSSE(toSSEFrame(ev))
          if (ev.type === "final" || ev.type === "error") {
            watchdog.cancel()                               // 정상 종료 — timer 영구 중단
          }
        }

        try {
          // UI-04 / D-13.4: per-query staleness check (cached 30s — Plan 05)
          const report: StalenessReport = await staleCheck(env)
          if (report.unknown.length > 0 || report.stale.length > 0) {
            await emit({
              type: "drift",
              message: `⚠ services.yaml 불일치: docker-only ${report.unknown.length}개, yaml-only ${report.stale.length}개 발견`,
              unknown: report.unknown,
              stale: report.stale,
            })
          }

          // KB-02: top-k=5 RAG retrieval
          const hits = await queryTopK(prompt, 5)
          const ragContext = formatRagContext(hits)

          // agent/loop.ts iterate — thin pump.
          // watchdog.signal을 iterate()에 전달 → 60s chunk timeout 시 loop가 abort.
          await iterate(prompt, {
            sessionId,
            ragContext,
            emit: (ev) => { void emit(ev) },
            abortSignal: watchdog.signal,                   // LOOP-06: chunk timeout → loop abort
          })
        } catch (err) {
          // 모든 throw가 이 catch로 — emit error 후 stream close.
          // watchdog이 이미 발화한 경우(watchdogFired)는 동일 envelope 재전송 회피.
          if ((err as Error).name !== "AbortError" && !watchdogFired) {
            await emit({
              type: "error",
              problem: "서버 처리 실패",
              cause: ((err as Error).message ?? String(err)).slice(0, 200),
              fix: "서버 로그 확인 후 재시도",
              retryable: true,
            })
          }
        } finally {
          watchdog.cancel()                                 // 모든 경로에서 timer 누수 방지
        }
      }, async (err, stream) => {
        // streamSSE error handler — PITFALL #4 AbortError 무시
        if ((err as Error).name !== "AbortError") {
          await stream.writeSSE({ event: "error", data: JSON.stringify({ problem: String(err), cause: "stream", fix: "재연결", retryable: true }) })
        }
      })
    })

    // ===== Startup probes =====
    async function startup(): Promise<void> {
      // probe 1: env (loadEnv 이미 호출됨 — 모듈 진입점에서 실패 시 process.exit)
      // probe 2: docker context
      const dockerCheck = await ensureDockerContext(env.DOCKER_CONTEXT)
      if (!dockerCheck.ok) {
        console.error(`[BOOT-04] ${dockerCheck.problem}`)
        if ("cause" in dockerCheck) console.error(`원인: ${(dockerCheck as { cause: string }).cause}`)
        if ("fix" in dockerCheck) console.error(`해결: ${(dockerCheck as { fix: string }).fix}`)
        process.exit(1)
      }
      // probe 3: kb 인덱싱 (D-13.3 lazy hash)
      try {
        const result = await ensureIndexed()
        console.log(`[KB] indexed=${result.indexed}, chunkCount=${result.chunkCount}`)
      } catch (err) {
        console.error(`[KB] 인덱싱 실패: ${(err as Error).message}`)
        process.exit(1)
      }
      // probe 4: boot 시 staleCheck → 결과 콘솔 출력 (drift event는 per-query에서)
      try {
        const report = await staleCheck(env)
        if (report.unknown.length > 0 || report.stale.length > 0) {
          console.warn(`[KB-03] services.yaml drift 감지: unknown=${report.unknown.length}, stale=${report.stale.length}`)
        }
      } catch (err) {
        console.warn(`[KB-03] staleCheck 실패 (서버 시작 계속): ${(err as Error).message}`)
      }
    }

    // 모듈이 직접 실행될 때만 startup + serve (test에서 import만 하면 server 시작 안 함)
    if (import.meta.main) {
      await startup()
      console.log(`[server] ready — http://127.0.0.1:${process.env.PORT ?? 3000}`)
    }

    // PITFALL #16: 127.0.0.1 bind (0.0.0.0 절대 금지)
    export default {
      hostname: "127.0.0.1",
      port: Number(process.env.PORT ?? 3000),
      fetch: app.fetch,
    }
    export { app }
    ```
